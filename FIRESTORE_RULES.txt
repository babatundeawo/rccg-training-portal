// Paste this into Firebase console > Firestore Database > Rules, then Publish.
// Deny by default: a signed-in person can only touch their own progress document.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isOwner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }

    // Progress may only hold: modules (map), updatedAt, email (string).
    function isValidProgress() {
      let d = request.resource.data;
      return d.keys().hasOnly(['modules', 'updatedAt', 'email'])
        && (!('modules' in d) || d.modules is map)
        && (!('updatedAt' in d) || d.updatedAt is number)
        && (!('email' in d) || d.email == null || (d.email is string && d.email.size() < 321));
    }

    match /progress/{uid} {
      allow get, delete: if isOwner(uid);
      allow create, update: if isOwner(uid) && isValidProgress();
      allow list: if false;
    }

    // Everything else is closed.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
