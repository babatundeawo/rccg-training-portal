// Paste this into Firebase console > Firestore Database > Rules, then Publish.
// Deny by default. What each collection is for:
//   progress/{uid}  A learner's study progress. Only they can write it. Class leaders
//                   can read it ONLY for learners who switched on "Share my progress".
//   private/{uid}   A learner's notes, bookmarks and reading positions. Only they can
//                   read or write it. Leaders can never read it.
//   leaders/{uid}   Marks who is an approved class leader. Nobody can write it from the
//                   website; the site owner adds a document by hand in the Firebase console.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isOwner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }

    // An approved leader has a verified email and a leaders/{their uid} document with active = true.
    function isLeader() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && get(/databases/$(database)/documents/leaders/$(request.auth.uid)).data.active == true;
    }

    // Progress may only hold these fields, with the right kinds of values.
    function isValidProgress() {
      let d = request.resource.data;
      return d.keys().hasOnly(['modules', 'activity', 'completed', 'updatedAt', 'email', 'name', 'shared'])
        && (!('modules' in d) || d.modules is map)
        && (!('activity' in d) || (d.activity is list && d.activity.size() <= 400))
        && (!('completed' in d) || d.completed is map)
        && (!('updatedAt' in d) || d.updatedAt is number)
        && (!('email' in d) || d.email == null || (d.email is string && d.email.size() < 321))
        && (!('name' in d) || (d.name is string && d.name.size() <= 80))
        && (!('shared' in d) || d.shared is bool);
    }

    function isValidPrivate() {
      let d = request.resource.data;
      return d.keys().hasOnly(['notes', 'bookmarks', 'positions', 'updatedAt'])
        && (!('notes' in d) || d.notes is map)
        && (!('bookmarks' in d) || d.bookmarks is map)
        && (!('positions' in d) || d.positions is map)
        && (!('updatedAt' in d) || d.updatedAt is number);
    }

    match /progress/{uid} {
      allow get: if isOwner(uid) || (isLeader() && resource.data.shared == true);
      // Leaders list only learners who opted in (the app asks for exactly that).
      allow list: if isLeader() && resource.data.shared == true;
      allow create, update: if isOwner(uid) && isValidProgress();
      allow delete: if isOwner(uid);
    }

    match /private/{uid} {
      allow get, delete: if isOwner(uid);
      allow create, update: if isOwner(uid) && isValidPrivate();
      allow list: if false;
    }

    match /leaders/{uid} {
      allow get: if isOwner(uid);
      allow list, write: if false;
    }

    // Everything else is closed.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
